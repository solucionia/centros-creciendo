                    console.log(`✅ Clic exitoso en botón: "${textoBoton}"`);
                    continuarExitoso = true;
                    break;
                  }
                } catch (error) {
                  console.log(
                    `   ⚪ Error inspeccionando elemento ${i}: ${error.message}`
                  );
                }
              }
              if (continuarExitoso) break;
            } catch (error) {
              console.log(
                `⚠️ Error con selector ${selectorBase}: ${error.message}`
              );
            }
          }

          // Si no encuentra el botón específico, intentar con el último botón visible
          if (!continuarExitoso) {
            console.log(
              "🔍 Intentando hacer clic en el último botón verde/visible..."
            );
            try {
              const ultimoBotón = await paginaDriCloud.$(
                "button:visible:last-of-type, .btn:visible:last-of-type"
              );
              if (ultimoBotón) {
                const textoUltimo = await ultimoBotón.textContent();
                console.log(
                  `🎯 Haciendo clic en último botón: "${textoUltimo}"`
                );
                await ultimoBotón.click();
                console.log("✅ Clic en último botón exitoso");
                continuarExitoso = true;
              }
            } catch (error) {
              console.log("⚠️ Error con último botón:", error.message);
            }
          }

          if (!continuarExitoso) {
            console.log(
              "❌ No se pudo encontrar botón para continuar desde fecha/hora"
            );
            console.log(
              '👆 Por favor, haz clic MANUALMENTE en el botón verde "Seleccionar"'
            );
            console.log("⏳ El script continuará cuando avances...");

            // Esperar a que el usuario haga clic manualmente
            try {
              await paginaDriCloud.waitForURL("**/*", { timeout: 30000 });
              console.log("✅ Usuario avanzó manualmente desde fecha/hora");
            } catch (error) {
              console.log("⏳ Continuando con la automatización...");
            }
          }
        } catch (error) {
          console.log("⚠️  Error seleccionando hora:", error.message);
        }
      } else {
        console.log("⚠️  No se pudo seleccionar ninguna fecha");
      }
    } catch (error) {
      console.log("⚠️  Error en selección de fecha/hora:", error.message);
    }

    await new Promise((resolve) => setTimeout(resolve, 4000));

    // PASO 5: Rellenar formulario del NUEVO paciente (SIN HACER CLIC EN BOTONES)
    console.log("📋 PASO 5: Rellenando formulario del NUEVO paciente...");
    console.log(
      "⚠️  IMPORTANTE: NO se hará clic en ningún botón después de rellenar"
    );

    try {
      console.log(
        "🔍 Buscando campos del formulario de nuevo paciente directamente..."
      );

      // Diagnóstico: Verificar si el formulario existe
      try {
        const formulario = await paginaDriCloud.$("form#formNuevoPaciente");
        if (formulario) {
          console.log("✅ Formulario 'formNuevoPaciente' encontrado");
        } else {
          console.log("❌ Formulario 'formNuevoPaciente' NO encontrado");
        }
      } catch (error) {
        console.log("⚠️ Error verificando formulario:", error.message);
      }

      // Rellenar cada campo con selectores basados en el HTML real
      console.log("🚀 Rellenando campos usando selectores exactos del HTML...");

      const camposEspecificos = [
        {
          valor: datosDelPaciente.nombre,
          selectores: ["input#PAC_NOMBRE", 'input[name="PAC_NOMBRE"]'],
          label: "Nombre (PAC_NOMBRE)",
        },
        {
          valor: datosDelPaciente.apellidos,
          selectores: ["input#PAC_APELLIDOS", 'input[name="PAC_APELLIDOS"]'],
          label: "Apellidos (PAC_APELLIDOS)",
        },
        {
          valor: datosDelPaciente.fechaNacimiento,
          selectores: [
            "input#datetimepicker1",
            'input[name="PAC_FECHA_NACIMIENTO"]',
          ],
          label: "Fecha de Nacimiento (datetimepicker1)",
        },
        {
          valor: datosDelPaciente.telefono,
          selectores: ["input#PAC_TELEFONO1", 'input[name="PAC_TELEFONO1"]'],
          label: "Teléfono (PAC_TELEFONO1)",
        },
        {
          valor: datosDelPaciente.dni,
          selectores: ["input#PAC_NIF", 'input[name="PAC_NIF"]'],
          label: "DNI/NIE/NIF (PAC_NIF)",
        },
        {
          valor: datosDelPaciente.pasaporte,
          selectores: ["input#PAC_PASAPORTE", 'input[name="PAC_PASAPORTE"]'],
          label: "Pasaporte (PAC_PASAPORTE)",
        },
        {
          valor: datosDelPaciente.email,
          selectores: ["input#PAC_EMAIL", 'input[name="PAC_EMAIL"]'],
          label: "Email (PAC_EMAIL)",
        },
      ];

      for (const campo of camposEspecificos) {
        if (campo.valor && campo.valor.trim()) {
          console.log(`   📝 Buscando ${campo.label}: "${campo.valor}"`);

          let campoRellenado = false;
          for (const selector of campo.selectores) {
            try {
              console.log(`     🔍 Probando selector: ${selector}`);
              const elemento = await paginaDriCloud.$(selector);

              if (elemento) {
                const esVisible = await elemento.isVisible();
                const esHabilitado = await elemento.isEnabled();

                console.log(
                  `       📊 Elemento encontrado - Visible: ${esVisible}, Habilitado: ${esHabilitado}`
                );

                if (esVisible && esHabilitado) {
                  console.log(`       📝 Rellenando con: "${campo.valor}"`);

                  // Limpiar campo y rellenar
                  await elemento.click({ clickCount: 3 });
                  await elemento.fill(campo.valor);

                  // Verificar que se rellenó
                  const valorActual = await elemento.inputValue();
                  console.log(
                    `       ✅ ${campo.label} rellenado correctamente. Valor actual: "${valorActual}"`
                  );

                  campoRellenado = true;
                  break;
                } else {
                  console.log(`       ⚪ Elemento no visible o deshabilitado`);
                }
              } else {
                console.log(
                  `       ❌ Elemento no encontrado con selector: ${selector}`
                );
              }
            } catch (error) {
              console.log(
                `       ⚠️ Error con selector ${selector}: ${error.message}`
              );
            }
          }

          if (!campoRellenado) {
            console.log(`     ❌ No se pudo rellenar: ${campo.label}`);
          } else {
            await new Promise((resolve) => setTimeout(resolve, 800));
          }
        } else {
          console.log(`   ⚪ Saltando ${campo.label} (sin datos)`);
        }
      }

      // Manejar género específico
      if (datosDelPaciente.genero && datosDelPaciente.genero.trim()) {
        console.log(`   📝 Seleccionando género: "${datosDelPaciente.genero}"`);

        const selectoresGenero = ["select#cmbSEX_ID", 'select[name="SEX_ID"]'];

        let generoSeleccionado = false;
        for (const selector of selectoresGenero) {
          try {
            console.log(`     🔍 Probando selector género: ${selector}`);
            const elemento = await paginaDriCloud.$(selector);

            if (elemento && (await elemento.isVisible())) {
              console.log(
                `       📝 Seleccionando género: "${datosDelPaciente.genero}"`
              );

              // Mapear géneros a los valores del HTML
              let valorGenero = "0"; // Sin determinar por defecto
              if (
                datosDelPaciente.genero.toLowerCase().includes("masculino") ||
                datosDelPaciente.genero.toLowerCase().includes("hombre") ||
                datosDelPaciente.genero.toLowerCase().includes("varón")
              ) {
                valorGenero = "2";
              } else if (
                datosDelPaciente.genero.toLowerCase().includes("femenino") ||
                datosDelPaciente.genero.toLowerCase().includes("mujer")
              ) {
                valorGenero = "1";
              }

              await elemento.selectOption({ value: valorGenero });
              console.log(
                `       ✅ Género seleccionado correctamente (valor: ${valorGenero})`
              );
              generoSeleccionado = true;
              break;
            }
          } catch (error) {
            console.log(
              `       ⚠️ Error con género ${selector}: ${error.message}`
            );
          }
        }

        if (!generoSeleccionado) {
          console.log(`     ❌ No se pudo seleccionar género`);
        }
      }

      // Seleccionar aseguradora si está disponible
      if (datosDelPaciente.aseguradora && datosDelPaciente.aseguradora.trim()) {
        console.log(
          `   📝 Seleccionando aseguradora: "${datosDelPaciente.aseguradora}"`
        );

        try {
          const elemento = await paginaDriCloud.$("select#SOC_ID");

          if (elemento && (await elemento.isVisible())) {
            // Mapear aseguradoras
            let valorAseguradora = "";
            const aseguradoraLower = datosDelPaciente.aseguradora.toLowerCase();

            if (
              aseguradoraLower.includes("fundación") ||
              aseguradoraLower.includes("fundacion")
            ) {
              valorAseguradora = "3";
            } else if (aseguradoraLower.includes("miembro")) {
              valorAseguradora = "2";
            } else if (
              aseguradoraLower.includes("privado") ||
              aseguradoraLower.includes("privada")
            ) {
              valorAseguradora = "1";
            }

            if (valorAseguradora) {
              await elemento.selectOption({ value: valorAseguradora });
              console.log(
                `     ✅ Aseguradora seleccionada (valor: ${valorAseguradora})`
              );
            } else {
              console.log(
                `     ⚪ Aseguradora "${datosDelPaciente.aseguradora}" no reconocida, dejando por defecto`
              );
            }
          }
        } catch (error) {
          console.log(`     ⚠️ Error con aseguradora: ${error.message}`);
        }
      }

      // Marcar checkbox de privacidad específico
      console.log("   📝 Marcando checkbox de privacidad...");

      try {
        const elemento = await paginaDriCloud.$(
          "input#chkAceptarPoliticaPrivacidad2"
        );

        if (elemento) {
          const esVisible = await elemento.isVisible();
          const esHabilitado = await elemento.isEnabled();
          const estaMarcado = await elemento.isChecked();

          console.log(
            `     📊 Checkbox - Visible: ${esVisible}, Habilitado: ${esHabilitado}, Marcado: ${estaMarcado}`
          );

          if (esVisible && esHabilitado && !estaMarcado) {
            console.log(`     ✅ Marcando checkbox de privacidad`);
            await elemento.check();
            console.log(`     ✅ Checkbox marcado correctamente`);
          } else if (estaMarcado) {
            console.log(`     ✅ Checkbox ya estaba marcado`);
          }
        } else {
          console.log(
            `     ❌ No se encontró checkbox con ID: chkAceptarPoliticaPrivacidad2`
          );
        }
      } catch (error) {
        console.log(`     ⚠️ Error con checkbox: ${error.message}`);
      }

      // Manejar género (select) - priorizando sección de nuevo paciente
      if (datosDelPaciente.genero) {
        console.log(`   📝 Seleccionando género: "${datosDelPaciente.genero}"`);
        try {
          const selectoresGenero = [
            'div:has-text("Detalles de nuevo paciente") select[name*="genero" i]',
            'div:has-text("nuevo paciente") select[name*="sexo" i]',
            'div:has-text("nuevo paciente") ~ * select[id*="genero" i]',
            'select[name*="genero" i]:nth-last-of-type(1)',
            'select[name*="sexo" i]:nth-last-of-type(1)',
            'select[id*="genero" i]:nth-last-of-type(1)',
            'select[name*="genero" i]',
            'select[name*="sexo" i]',
            'select[id*="genero" i]',
          ];

          let generoSeleccionado = false;
          for (const selector of selectoresGenero) {
            try {
              const elementos = await paginaDriCloud.$$(selector);

              // Priorizar último select encontrado (nuevo paciente)
              for (let i = elementos.length - 1; i >= 0; i--) {
                const elemento = elementos[i];
                if (elemento && (await elemento.isVisible())) {
                  await elemento.selectOption({
                    label: datosDelPaciente.genero,
                  });
                  console.log(
                    `     ✅ Género seleccionado: ${selector} (elemento ${
                      i + 1
                    })`
                  );
                  generoSeleccionado = true;
                  break;
                }
              }

              if (generoSeleccionado) break;
            } catch (error) {
              console.log(`     ⚪ Género no disponible: ${selector}`);
            }
          }

          if (!generoSeleccionado) {
            console.log("     ❌ No se pudo seleccionar género");
          }
        } catch (error) {
          console.log("   ⚠️  Error seleccionando género:", error.message);
        }
      }

      // Marcar checkbox de política de privacidad - priorizando sección de nuevo paciente
      console.log("   📝 Marcando política de privacidad...");
      try {
        const selectoresCheckbox = [
          'div:has-text("Detalles de nuevo paciente") input[type="checkbox"]:not(:checked)',
          'div:has-text("nuevo paciente") input[name*="privacidad" i]:not(:checked)',
          'div:has-text("nuevo paciente") ~ * input[name*="politica" i]:not(:checked)',
          'div:has-text("nuevo paciente") ~ * input[name*="acepto" i]:not(:checked)',
          'input[type="checkbox"]:not(:checked):nth-last-of-type(1)',
          'input[type="checkbox"]:not(:checked):nth-last-of-type(2)',
          'input[name*="privacidad" i]:not(:checked):nth-last-of-type(1)',
          'input[name*="politica" i]:not(:checked):nth-last-of-type(1)',
          'input[name*="acepto" i]:not(:checked):nth-last-of-type(1)',
          'input[name*="terminos" i]:not(:checked):nth-last-of-type(1)',
          'input[type="checkbox"]:not(:checked)',
          'input[name*="privacidad" i]:not(:checked)',
          'input[name*="politica" i]:not(:checked)',
          'input[name*="acepto" i]:not(:checked)',
          'input[name*="terminos" i]:not(:checked)',
        ];

        let checkboxMarcado = false;
        for (const selector of selectoresCheckbox) {
          try {
            const elementos = await paginaDriCloud.$$(selector);

            // Priorizar último checkbox encontrado (nuevo paciente)
            for (let i = elementos.length - 1; i >= 0; i--) {
              const elemento = elementos[i];
              try {
                const esVisible = await elemento.isVisible();
                const esHabilitado = await elemento.isEnabled();

                if (esVisible && esHabilitado) {
                  await elemento.check();
                  console.log(
                    `     ✅ Checkbox marcado: ${selector} (elemento ${i + 1})`
                  );
                  checkboxMarcado = true;
                  break;
                }
              } catch (error) {
                console.log(
                  `     ⚪ Error checkbox elemento ${i + 1}: ${error.message}`
                );
              }
            }

            if (checkboxMarcado) break;
          } catch (error) {
            console.log(`     ⚪ Checkbox no disponible: ${selector}`);
          }
        }

        if (!checkboxMarcado) {
          console.log("     ⚠️  No se encontró checkbox de privacidad");
        }
      } catch (error) {
        console.log("   ⚠️  Error marcando checkbox:", error.message);
      }

      console.log("");
      console.log("✅ Formulario de NUEVO paciente completado");
      console.log(
        "🔍 Los datos se han rellenado priorizando la sección inferior del formulario"
      );
    } catch (error) {
      console.log(
        "⚠️  Error rellenando formulario del nuevo paciente:",
        error.message
      );
    }

    // Actualizar notificación de estado
    await paginaDriCloud.evaluate(() => {
      const notif = document.getElementById("estado-automatizacion");
      if (notif) {
        notif.innerHTML =
          "✅ FORMULARIO RELLENADO<br><small>NO se presionará ningún botón automáticamente</small>";
        notif.style.background = "linear-gradient(45deg, #27ae60, #2ecc71)";
      }
    });

    // Capturas finales
    console.log("");
    console.log("📸 Generando capturas del resultado...");
    try {
      await paginaFormulario.screenshot({
        path: "formulario-datos-enviados.png",
        fullPage: true,
      });
      await paginaDriCloud.screenshot({
        path: "dricloud-formulario-completado.png",
        fullPage: true,
      });
      console.log("✅ Capturas guardadas correctamente");
    } catch (error) {
      console.log("⚠️  Error generando capturas:", error.message);
    }

    console.log("");
    console.log("🎊 ¡FORMULARIO RELLENADO COMPLETAMENTE!");
    console.log("======================================");
    console.log("");
    console.log("📋 Proceso realizado:");
    console.log("   ✅ Formulario de datos procesado correctamente");
    console.log("   ✅ Conexión a DriCloud establecida");
    console.log("   ✅ Servicio médico seleccionado");
    console.log("   ✅ Profesional (Dr. Fernando) seleccionado");
    console.log("   ✅ Fecha y hora configuradas");
    console.log("   ✅ Formulario del NUEVO paciente rellenado");
    console.log("");
    console.log("🔒 IMPORTANTE - EL SCRIPT SE DETIENE AQUÍ:");
    console.log("   ✅ NO se ha hecho clic en ningún botón de continuar");
    console.log("   🛑 NO se ha hecho clic en el botón 'Continuar'");
    console.log("   🔍 Los datos están rellenados pero NO enviados");
    console.log("   👆 Puedes revisar y decidir manualmente qué hacer");
    console.log("");
    console.log("📷 Archivos generados:");
    console.log("   - formulario-datos-enviados.png (tus datos introducidos)");
    console.log(
      "   - dricloud-formulario-completado.png (formulario rellenado)"
    );
    console.log("");
    console.log("🔄 VENTANAS PERMANENTES:");
    console.log("   ⚠️  Las ventanas permanecerán ABIERTAS");
    console.log("   👆 Puedes revisar y decidir si reservar manualmente");
    console.log("   🔚 Cierra el navegador cuando termines de revisar");

    // Mantener abierto hasta que se cierre el navegador manualmente
    console.log("");
    console.log(
      "🔄 Modo permanente activado - Esperando hasta que cierres el navegador..."
    );

    await new Promise((resolve) => {
      browser.on("disconnected", () => {
        console.log("🔚 Navegador cerrado por el usuario - Script finalizado");
        resolve();
      });
    });
  } catch (error) {
    console.error("❌ Error general en la automatización:", error.message);
    console.log("");
    console.log("🔄 Manteniendo ventanas abiertas a pesar del error...");

    try {
      await new Promise((resolve) => {
        browser.on("disconnected", () => {
          console.log("🔚 Script finalizado");
          resolve();
        });
      });
    } catch (finalError) {
      console.log("🔚 Finalizando script");
    }
  }
}
